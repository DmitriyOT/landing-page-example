    47→function AnimatedSection({ children, className = "" }: { children: React.ReactNode; className?: string }) {
    48→  const ref = useRef(null);
    49→  const isInView = useInView(ref, { once: true, margin: "-80px" });
    50→  return (
    51→    <motion.div
    52→      ref={ref}
    53→      initial="hidden"
    54→      animate={isInView ? "visible" : "hidden"}
    55→      className={className}
    56→    >
    57→      {children}
    58→    </motion.div>
    59→  );
    60→}
    61→
    62→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
    63→  const ref = useRef(null);
    64→  const isInView = useInView(ref, { once: true });
    65→  const count = useMotionValue(0);
    66→  const rounded = useTransform(count, (v) => {
    67→    if (target >= 1000) return `${Math.round(v / 1000)}K+`;
    68→    if (suffix === "+") return `${Math.round(v)}+`;
    69→    if (suffix === "B") return `$${(v / 1).toFixed(0)}B`;
    70→    return `${Math.round(v)}`;
    71→  });
    72→
    73→  useEffect(() => {
    74→    if (!isInView) return;
    75→    const controls = animate(count, target, {
    76→      duration: 2.2,
    77→      ease: [0.22, 1, 0.36, 1],
    78→    });
    79→    return () => controls.stop();
    80→  }, [isInView, target, count]);
    81→
    82→  return <motion.span ref={ref}>{rounded}</motion.span>;
    83→}
