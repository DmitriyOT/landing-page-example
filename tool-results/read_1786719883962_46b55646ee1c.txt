    47→    47→function AnimatedSection({ children, className = "" }: { children: React.ReactNode; className?: string }) {
    48→    48→  const ref = useRef(null);

    62→    62→function AnimatedCounter({ target, suffix = "" }: { target: number; suffix?: string }) {
    63→    63→  const ref = useRef(null);
    64→    64→  const isInView = useInView(ref, { once: true });
    65→    65→  const count = useMotionValue(0);
    66→    66→  const rounded = useTransform(count, (v) => {
    67→    67→    if (target >= 1000) return `${Math.round(v / 1000)}K+`;
    68→    68→    if (suffix === "+") return `${Math.round(v)}+`;
    69→    69→    if (suffix === "B") return `$${(v / 1).toFixed(0)}B`;
    70→    70→    return `${Math.round(v)}`;
    71→    71→  });
    72→    72→
    73→    73→  useEffect(() => {
    74→    74→    if (!isInView) return;
    75→    75→    const controls = animate(count, target, {
    76→    76→      duration: 2.2,
    77→    77→      ease: [0.22, 1, 0.36, 1],
    78→    78→    });
    79→    79→    return () => controls.stop();
    80→    80→  }, [isInView, target, count]);
    81→    81→
    82→    82→  return <motion.span ref={ref}>{rounded}</motion.span>;
    83→    83→}
